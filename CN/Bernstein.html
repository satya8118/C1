# ----- Table 1: numerical values at x = 0.3 -----
from math import comb

def B(n, k, x):
    return comb(n, k) * (x ** k) * ((1 - x) ** (n - k))

x_val = 0.3
print("Table 1: B_{k,n}(0.3)\n")
print("      " + "".join(f"   k={k} " for k in range(6)))
for n in range(1, 7):
    row = f"n={n} "
    for k in range(n + 1):
        row += f" {B(n, k, x_val):.4f} "
    print(row)

# ----- Table 2: symbolic polynomials B_{k,n}(x) -----
def poly_term(n, k):
    if k == 0:
        return f"(1-x)^{n}"
    elif k == n:
        return f"x^{n}"
    else:
        return f"x^{k}(1-x)^{n-k}"

print("\nTable 2: B_{k,n}(x) = binom(n,k) · x^k · (1-x)^(n-k)\n")
print("      " + "".join(f"   k={k} " for k in range(6)))
for n in range(1, 7):
    row = f"n={n} "
    for k in range(n + 1):
        coeff = comb(n, k)
        term = poly_term(n, k)
        row += f" {coeff}·{term} "
    print(row)
